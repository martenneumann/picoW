interface steuerung {
  .. oeffentliche Funktionen ..
  getEntfernungsArray(AktuelleTemperatur)
  getArithmetischesMittelFromArray(messungs_Array)
  
  .. globale Variablen ..
  temperaturMessungenNachXms
  temperaturNachXmessungenErneuern
  anzahlMessungenProSample
  i 'Zähler für Messdurchläufe'
  logStrategie
  
  .. private Funktionen ..
  getEntfernungsArray(aktuelleTemperatur)
  getEntfernungsDataFromMessung(aktuelleTemperatur)
  getArithmetischesMittelFromArray(myArray)
  getVarianzFromArray(myArray, aritMittel)
  getStandartAbweichungFromArray(variant_Array)
  getStandartAbweichungDesMittelwerts(stdAbweichung)
  get2SigmaFromStdAbweichung(myStdAbweichung)
  getSchallgeschwindigkeit(akruelleTemperatur)
  getSpeedArray(messungs_Array, t)
  getSpeed(d1, d2, t)
  tonlaufzeit2Strecke(tonLaufzeit, aktuelleTemperatur)
  main()
}

}

  dumpKalibrierungsMessungAlleWerte(messungs_Array)
  dumpInit(Kalibrierungspunkte_User, kalPunktEins, messungsKorrekturFaktro)
